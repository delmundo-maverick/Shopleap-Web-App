namespace App\Http\Controllers\Buyer;

class HomeController extends Controller
{
    public function index(Request $request, PublicCatalog $catalog)
    {
        $categories = config('shopleap_categories.list');

        $bestSellers = $catalog->featured();

        $recommended = $catalog->query()->paginate(16)->withQueryString();

        $cartItems = $request->user()->cartItems()

            ->get();

            'cartCount',
            'cartPreviewItems'
        ));
    }
}

<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Models\BuyerProfile;
use App\Models\LogisticsProfile;
use App\Models\SellerProfile;
use App\Models\User;
use App\Services\OtpService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;

class LoginController extends Controller
{
    protected array $validRoles = ['buyer', 'seller', 'logistics'];

    protected function checkRole(string $role): void
    {
        abort_unless(in_array($role, $this->validRoles), 404);
    }

    // â”€â”€ LOGIN â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    public function create(Request $request)
    {
        if ($request->query('continue') === 'landing') {
            $filters = $request->validate([
                'search' => ['nullable', 'string', 'max:100'],
                'category' => ['nullable', 'string', 'max:100'],
            ]);
            $request->session()->put('url.intended', route('home', $filters));
            $request->session()->flash('status', 'Please log in to continue');
        } elseif ($request->session()->has('url.intended')) {
            $request->session()->flash('status', 'Please log in to continue');
        }

        return view('auth.login');
    }

    public function store(Request $request)
    {
        $credentials = $request->validate([
            'email' => ['required', 'email'],
            'password' => ['required'],
        ]);

        if (! Auth::attempt($credentials, $request->boolean('remember'))) {
            return back()->withErrors([
                'email' => 'These credentials do not match our records.',
            ])->onlyInput('email');
        }

        $user = Auth::user();

        if (is_null($user->email_verified_at)) {
            Auth::logout();
            $request->session()->invalidate();
            $request->session()->regenerateToken();

            return back()->withErrors([
                'email' => 'Please verify your email before logging in.',
            ])->onlyInput('email');
        }

        // Couriers must sign in through the Shopleap Rider app.
        if ($user->role === 'courier') {
            Auth::logout();
            $request->session()->invalidate();
            $request->session()->regenerateToken();

            return back()->withErrors([
                'email' => 'Riders sign in through the Shopleap Rider app.',
            ])->onlyInput('email');
        }

        $profile = match ($user->role) {
            'seller' => $user->sellerProfile,
            'buyer' => $user->buyerProfile,
            'logistics' => $user->logisticsProfile,
            default => null,
        };

        if ($profile && $profile->status !== 'approved') {
            Auth::logout();
            $request->session()->invalidate();
            $request->session()->regenerateToken();

            $message = $profile->status === 'rejected'
                ? 'Your registration was not approved. Please contact support for details.'
                : "Your registration is still pending administrator approval. You'll be notified by email once reviewed.";

            return back()->withErrors([
                'email' => $message,
            ])->onlyInput('email');
        }

        if ($user->account_status !== 'active') {
            Auth::logout();
            $request->session()->invalidate();
            $request->session()->regenerateToken();

            $message = $user->account_status === 'suspended'
                ? 'Your account has been suspended. Please contact support.'
                : 'Your account has been deactivated.';

            return back()->withErrors([
                'email' => $message,
            ])->onlyInput('email');
        }

        $request->session()->regenerate();

        return match ($user->role) {
            'super_admin' => redirect()->intended(route('admin.dashboard')),
            'seller' => redirect()->intended(route('seller.dashboard')),
            'buyer' => redirect()->intended(route('buyer.home')),
            'logistics' => redirect()->intended(route('logistics.dashboard')),
            default => redirect()->intended(route('home')),
        };
    }

    public function destroy(Request $request)
    {
        Auth::guard('web')->logout();

        $request->session()->invalidate();
        $request->session()->regenerateToken();

        return redirect()->route('login');
    }

    // â”€â”€ REGISTRATION STEP 2: OTP VERIFY â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    public function showOtpForm(string $role)
    {
        $this->checkRole($role);

        $pending = session('pending_registration');

        abort_if(
            ! $pending || $pending['role'] !== $role,
            403,
            'Please start registration again.'
        );

        return view('auth.verify-otp', [
            'role' => $role,
            'email' => $pending['email'],
        ]);
    }

    public function verifyOtp(
        Request $request,
        string $role,
        OtpService $otpService
    ) {
        $this->checkRole($role);

        $request->validate([
            'otp' => 'required|digits:6',
        ]);

        $pending = session('pending_registration');

        abort_if(
            ! $pending || $pending['role'] !== $role,
            403,
            'Please start registration again.'
        );

        [$ok, $message] = $otpService->verify(
            $pending['email'],
            $role,
            $request->otp
        );

        if (! $ok) {
            if ($request->wantsJson()) {
                return response()->json([
                    'message' => $message,
                ], 422);
            }

            return back()->withErrors([
                'otp' => $message,
            ]);
        }

        DB::transaction(function () use ($pending, $role) {
            $user = User::create(
                $pending['user'] + ['email_verified_at' => now()]
            );

            $profileData = $pending['profile'] + [
                'user_id' => $user->id,
            ];

            match ($role) {
                'buyer' => BuyerProfile::create($profileData),
                'seller' => SellerProfile::create($profileData),
                'logistics' => LogisticsProfile::create($profileData),
            };
        });

        session()->forget('pending_registration');

        if ($request->wantsJson()) {
            return response()->json([
                'message' => 'verified',
            ]);
        }

        return redirect()->route('login')->with(
            'status',
            "Email verified and registration submitted! Please wait for the administrator's approval â€” you'll be notified by email."
        );
    }

    public function resendOtp(
        Request $request,
        string $role,
        OtpService $otpService
    ) {
        $this->checkRole($role);

        $pending = session('pending_registration');

        abort_if(
            ! $pending || $pending['role'] !== $role,
            403,
            'Please start registration again.'
        );

        $otpService->generateAndSend($pending['email'], $role);

        if ($request->wantsJson()) {
            return response()->json([
                'message' => 'A new code has been sent.',
            ]);
        }

        return back()->with('status', 'A new code has been sent.');
    }
}

<?php

namespace App\Services;

use App\Models\Product;
use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Collection;
use Illuminate\Support\Facades\Cache;

class PublicCatalog
{
    public function query(): Builder
    {
        return Product::active()->with(['images', 'seller.sellerProfile', 'liveDiscount', 'variants'])->latest();
    }

    public function featured(): Collection
    {
        return Cache::remember('public-catalog.featured.v1', 60, fn (): Collection => $this->query()->take(8)->get());
    }

    public function discover(string $search, string $category): Collection
    {
        $key = 'public-catalog.discover.v1.'.hash('sha256', json_encode([$search, $category]));

        return Cache::remember($key, 60, fn (): Collection => $this->query()
            ->when($search !== '', fn (Builder $query): Builder => $query->where('name', 'like', '%'.$search.'%'))
            ->when($category !== '', fn (Builder $query): Builder => $query->where('category', $category))
            ->take(16)->get());
    }
}

let registered = false;

export function registerLandingEffects(Alpine) {
    if (registered || !document.body.classList.contains('landing-page')) { return; }
    registered = true;

    const motion = window.matchMedia('(prefers-reduced-motion: reduce)');
    const pending = new Set();
    const timers = new Map();
    let observer;

    function finish(element) {
        pending.delete(element);
        observer?.unobserve(element);
        clearTimeout(timers.get(element));
        timers.delete(element);
        element.classList.remove('reveal-pending', 'reveal-animating');
        element.style.removeProperty('--reveal-delay');
    }

    function show(element) {
        element.classList.add('reveal-animating');
        element.classList.remove('reveal-pending');
        observer?.unobserve(element);
        timers.set(element, setTimeout(() => finish(element), 650));
    }

    try {
        if ('IntersectionObserver' in window) {
            observer = new IntersectionObserver(entries => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) { show(entry.target); }
                });
            }, { threshold: 0.05 });
            document.documentElement.classList.add('js-ready');
        }
    } catch {
        // Progressive enhancement: unsupported observers leave all content visible.
    }

    Alpine.directive('reveal', (element, { expression }, { cleanup }) => {
        if (!observer || motion.matches) { return; }
        const bounds = element.getBoundingClientRect();
        if (bounds.top < window.innerHeight && bounds.bottom > 0) { return; }

        const columns = getComputedStyle(element.parentElement).gridTemplateColumns.split(' ').length;
        const delay = (Number(expression) || 0) % Math.max(1, columns) * 60;
        element.style.setProperty('--reveal-delay', `${delay}ms`);
        observer.observe(element);
        pending.add(element);
        element.classList.add('reveal-pending');
        const focus = () => finish(element);
        element.addEventListener('focusin', focus, { once: true });
        cleanup(() => {
            finish(element);
            element.removeEventListener('focusin', focus);
        });
    });

    motion.addEventListener('change', () => {
        if (motion.matches) { [...pending].forEach(finish); }
    });

    const header = document.querySelector('.landing-navigation');
    if (header) {
        let scheduled = false;
        const updateHeader = () => {
            header.classList.toggle('is-scrolled', window.scrollY > 8);
            scheduled = false;
        };
        updateHeader();
        window.addEventListener('scroll', () => {
            if (!scheduled) { scheduled = true; requestAnimationFrame(updateHeader); }
        }, { passive: true });
    }
}

Alpine.data('otpRegistration', (config = {}) => ({
    ...termsAgreementState(config),

    verifyUrl: config.verifyUrl,
    resendUrl: config.resendUrl,
    loginUrl: config.loginUrl,

    init() {
        this.initTermsAgreement();
    },

    submitting: false,
    showOtpModal: false,
    otpDigits: ['', '', '', '', '', ''],
    otpError: '',
    formErrors: [],
    email: '',
    verifying: false,
    resending: false,
    resendCooldown: 0,
    resendTimer: null,

    csrfToken() {
        return document.querySelector('meta[name="csrf-token"]').getAttribute('content');
    },

    async submitRegistration(event) {
        if (!this.termsAccepted) {
            this.openTermsModal();
            this.termsError =
                'Please read and agree to the Terms and Conditions and Privacy Policy before continuing.';
            return;
        }

        this.termsError = '';
        this.submitting = true;
        this.formErrors = [];

        const form = event.target;
        const formData = new FormData(form);

        try {
            const response = await fetch(form.action, {
                method: 'POST',
                headers: {
                    'X-Requested-With': 'XMLHttpRequest',
                    'X-CSRF-TOKEN': this.csrfToken(),
                    'Accept': 'application/json',
                },
                body: formData,
            });

            const data = await response.json();

            if (response.status === 422) {
                const errors = data.errors ?? {};
                this.formErrors = Object.values(errors).flat();

                if (errors.terms_accepted?.length) {
                    this.termsError = errors.terms_accepted[0];
                    this.openTermsModal();
                }

                window.scrollTo({ top: 0, behavior: 'smooth' });
                return;
            }

            if (!response.ok) {
                this.formErrors = [data.message || 'Something went wrong. Please try again.'];
                return;
            }

            this.email = data.email;
            this.openOtpModal();
        } catch (error) {
            console.error(error);
            this.formErrors = ['Network error. Please check your connection and try again.'];
        } finally {
            this.submitting = false;
        }
    },

    openOtpModal() {
        this.otpDigits = ['', '', '', '', '', ''];
        this.otpError = '';
        this.showOtpModal = true;
        this.startCooldown();
        this.$nextTick(() => this.$refs.otpInput0?.focus());
    },

    closeOtpModal() {
        this.showOtpModal = false;
        clearInterval(this.resendTimer);
    },

    handleOtpInput(index, event) {
        const digit = event.target.value.replace(/[^0-9]/g, '').slice(0, 1);
        this.otpDigits[index] = digit;
        event.target.value = digit;

        if (digit && index < 5) {
            this.$refs['otpInput' + (index + 1)]?.focus();
        }

        if (this.otpDigits.every((d) => d !== '')) {
            this.verifyOtp();
        }
    },

    handleOtpKeydown(index, event) {
        if (event.key === 'Backspace' && !this.otpDigits[index] && index > 0) {
            this.$refs['otpInput' + (index - 1)]?.focus();
        }
    },

    handleOtpPaste(event) {
        const pasted = (event.clipboardData?.getData('text') || '').replace(/[^0-9]/g, '').slice(0, 6);
        if (!pasted) return;
        event.preventDefault();

        pasted.split('').forEach((digit, i) => {
            this.otpDigits[i] = digit;
            if (this.$refs['otpInput' + i]) this.$refs['otpInput' + i].value = digit;
        });

        if (pasted.length === 6) {
            this.verifyOtp();
        } else {
            this.$refs['otpInput' + pasted.length]?.focus();
        }
    },

    async verifyOtp() {
        const code = this.otpDigits.join('');
        if (code.length !== 6 || this.verifying) return;

        this.verifying = true;
        this.otpError = '';

        try {
            const response = await fetch(this.verifyUrl, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'X-Requested-With': 'XMLHttpRequest',
                    'X-CSRF-TOKEN': this.csrfToken(),
                    'Accept': 'application/json',
                },
                body: JSON.stringify({ otp: code }),
            });

            const data = await response.json();

            if (!response.ok) {
                this.otpError = data.message || data.errors?.otp?.[0] || 'Invalid code. Please try again.';
                this.otpDigits = ['', '', '', '', '', ''];
                this.$nextTick(() => this.$refs.otpInput0?.focus());
                return;
            }

            window.location.href = this.loginUrl;
        } catch (error) {
            console.error(error);
            this.otpError = 'Network error. Please try again.';
        } finally {
            this.verifying = false;
        }
    },

    startCooldown() {
        this.resendCooldown = 60;
        clearInterval(this.resendTimer);
        this.resendTimer = setInterval(() => {
            this.resendCooldown--;
            if (this.resendCooldown <= 0) clearInterval(this.resendTimer);
        }, 1000);
    },

    async resendOtp() {
        if (this.resendCooldown > 0 || this.resending) return;

        this.resending = true;
        this.otpError = '';

        try {
            const response = await fetch(this.resendUrl, {
                method: 'POST',
                headers: {
                    'X-Requested-With': 'XMLHttpRequest',
                    'X-CSRF-TOKEN': this.csrfToken(),
                    'Accept': 'application/json',
                },
            });

            const data = await response.json();

            if (!response.ok) {
                this.otpError = data.message || 'Could not resend code. Please try again.';
                return;
            }

            this.startCooldown();
        } catch (error) {
            console.error(error);
            this.otpError = 'Network error. Please try again.';
        } finally {
            this.resending = false;
        }
    },
}));

Alpine.store('wishlist', {
    ids: {},          // { [productId]: true }
    count: 0,
    pending: {},      // { [productId]: true } while a request is in flight
    url: '',
    loginUrl: '/login',

    // Alpine calls init() automatically when the store is registered.
    init() {
        const el = document.getElementById('wishlistData');
        if (el) {
            const data = JSON.parse(el.textContent || '{}');
            (data.ids || []).forEach((id) => (this.ids[id] = true));
            this.count = data.count ?? 0;
            this.url = data.url;
            this.loginUrl = data.loginUrl || '/login';
        }

        // Back/forward cache can show a stale heart; refresh when restored.
        window.addEventListener('pageshow', (e) => {
            if (e.persisted) window.location.reload();
        });
    },

    has(id) {
        return !!this.ids[id];
    },

    set(id, on) {
        if (on) this.ids[id] = true;
        else delete this.ids[id];
    },

    headers(json = false) {
        return {
            Accept: 'application/json',
            'X-Requested-With': 'XMLHttpRequest',
            'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]')?.content,
            ...(json && { 'Content-Type': 'application/json' }),
        };
    },

    toast(message, error = false) {
        window.dispatchEvent(new CustomEvent('wishlist-toast', { detail: { message, error } }));
    },

    /** Optimistic toggle: update the UI now, roll back if the server says no. */
    async toggle(id) {
        if (this.pending[id]) return;

        const wasOn = this.has(id);
        this.pending[id] = true;
        this.set(id, !wasOn);
        this.count += wasOn ? -1 : 1;

        try {
            const res = await fetch(this.url.replace('__ID__', id), {
                method: wasOn ? 'DELETE' : 'POST',
                headers: this.headers(),
            });

            if (res.status === 401 || res.status === 419) {
                window.location.href = this.loginUrl;
                return;
            }

            const data = await res.json().catch(() => ({}));
            if (!res.ok || data.success === false) {
                throw new Error(data.message || 'Could not update your wishlist.');
            }

            this.count = data.wishlist_count; // trust the server's count
            this.toast(data.message);
        } catch (e) {
            this.set(id, wasOn);
            this.count += wasOn ? 1 : -1;
            this.toast(e instanceof TypeError ? 'Network error. Please try again.' : e.message, true);
        } finally {
            delete this.pending[id];
        }
    },

    /** Used by the wishlist page. Calls your existing cart endpoint. */
    async addToCart(url) {
        try {
            const res = await fetch(url, {
                method: 'POST',
                headers: this.headers(true),
                body: JSON.stringify({ quantity: 1 }),
            });
            const data = await res.json().catch(() => ({}));

            if (!res.ok || data.success === false) {
                this.toast(data.message || 'Could not add to cart.', true);
                return;
            }

            this.toast(data.message);
            const badge = document.getElementById('buyerCartBadge');
            if (badge && data.cart_count !== undefined) badge.textContent = data.cart_count;
        } catch {
            this.toast('Network error. Please try again.', true);
        }
    },
});

Alpine.store('follow', {
    ids: {},          // { [sellerId]: true }
    counts: {},       // { [sellerId]: follower count }
    total: 0,         // sellers this buyer follows
    pending: {},      // { [sellerId]: true } while a request is in flight
    followUrl: '',
    unfollowUrl: '',
    loginUrl: '/login',

    init() {
        const el = document.getElementById('sellerFollowData');
        if (!el) return;

        const data = JSON.parse(el.textContent || '{}');
        (data.ids || []).forEach((id) => (this.ids[id] = true));
        Object.entries(data.counts || {}).forEach(([id, count]) => {
            this.counts[id] = Number(count);
        });

        this.total = data.total ?? 0;
        this.followUrl = data.followUrl;
        this.unfollowUrl = data.unfollowUrl;
        this.loginUrl = data.loginUrl || '/login';
    },

    has(id) {
        return !!this.ids[id];
    },

    followerCount(id) {
        return this.counts[id] ?? 0;
    },

    /** Seed a seller's state on pages that show a single shop. */
    prime(id, count, following) {
        if (this.pending[id]) return;

        if (following) this.ids[id] = true;
        else delete this.ids[id];

        if (count !== null && count !== undefined) this.counts[id] = Number(count);
    },

    set(id, following) {
        if (following) this.ids[id] = true;
        else delete this.ids[id];
    },

    headers() {
        return {
            Accept: 'application/json',
            'X-Requested-With': 'XMLHttpRequest',
            'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]')?.content,
        };
    },

    toast(message, error = false) {
        window.dispatchEvent(new CustomEvent('seller-follow-toast', { detail: { message, error } }));
    },

    /** Optimistic toggle: update the UI now, roll back if the server says no. */
    async toggle(id) {
        if (this.pending[id]) return;

        const wasFollowing = this.has(id);
        const previousCount = this.followerCount(id);
        const previousTotal = this.total;

        this.pending[id] = true;
        this.set(id, !wasFollowing);
        this.counts[id] = Math.max(0, previousCount + (wasFollowing ? -1 : 1));
        this.total = Math.max(0, previousTotal + (wasFollowing ? -1 : 1));

        try {
            const res = await fetch((wasFollowing ? this.unfollowUrl : this.followUrl).replace('__ID__', id), {
                method: wasFollowing ? 'DELETE' : 'POST',
                headers: this.headers(),
            });

            if (res.status === 401 || res.status === 419) {
                window.location.href = this.loginUrl;
                return;
            }

            const data = await res.json().catch(() => ({}));
            if (!res.ok || data.success === false) {
                throw new Error(data.message || 'Could not update your following list.');
            }

            this.set(id, data.following);
            this.counts[id] = data.followers_count;
            this.total = data.following_count;
            this.toast(data.message);
        } catch (e) {
            this.set(id, wasFollowing);
            this.counts[id] = previousCount;
            this.total = previousTotal;
            this.toast(e instanceof TypeError ? 'Network error. Please try again.' : e.message, true);
        } finally {
            delete this.pending[id];
        }
    },
});

const locks = new Set();
let previousOverflow = '';
let previousBodyOverflow = '';

function scrollLock(id, enabled) {
    if (!enabled && !locks.has(id)) { return; }
    if (enabled && !locks.size) {
        previousOverflow = document.documentElement.style.overflow;
        previousBodyOverflow = document.body.style.overflow;
    }
    if (enabled) { locks.add(id); } else { locks.delete(id); }
    document.documentElement.style.overflow = locks.size ? 'hidden' : previousOverflow;
    document.body.style.overflow = locks.size ? 'hidden' : previousBodyOverflow;
}

export function navbar(config = {}) {
    return {
        id: config.id, desktop: window.innerWidth >= 1024, open: false, collapsed: false, activeDropdown: null, logoutOpen: false,
        logoutSubmitting: false, logoutTrigger: null, logoutFromMenu: false, returnFocus: null, dropdownTrigger: null, unread: 0, poll: null,
        init() {
            try { this.collapsed = localStorage.getItem(`navigation:${this.id}:collapsed`) === 'true'; } catch {}
            this.$watch('open', () => this.syncLock());
            this.$watch('logoutOpen', () => this.syncLock());
            if (config.unreadUrl) {
                this.refreshUnread();
                this.poll = setInterval(() => this.refreshUnread(), 20000);
            }
        },
        destroy() { clearInterval(this.poll); scrollLock(this.id, false); },
        syncLock() { scrollLock(this.id, this.logoutOpen || (this.open && window.innerWidth < 1024)); },
        toggle(trigger = null) {
            if (this.logoutOpen) { return; }
            if (window.innerWidth >= 1024) { return; }
            if (this.open) { this.close(); return; }
            this.returnFocus = trigger || document.activeElement;
            this.activeDropdown = null; this.open = true;
            this.broadcast();
            this.$nextTick(() => this.$refs.panel?.querySelector('button, a[href], input')?.focus());
        },
        close(restoreFocus = true) {
            const wasOpen = this.open;
            this.open = false; this.activeDropdown = null; this.syncLock(); this.broadcast();
            if (wasOpen && restoreFocus) { this.returnFocus?.focus(); }
        },
        broadcast() { window.dispatchEvent(new CustomEvent('navigation-state', { detail: { id: this.id, open: this.open } })); },
        resize() { this.desktop = window.innerWidth >= 1024; if (this.desktop) { this.close(false); } },
        collapse() {
            this.collapsed = !this.collapsed;
            try { localStorage.setItem(`navigation:${this.id}:collapsed`, String(this.collapsed)); } catch {}
        },
        navigate(event) { if (event.target.closest('a[href]')) { this.close(false); } },
        escape() {
            if (this.logoutOpen) { this.cancelLogout(); return; }
            if (this.activeDropdown) { this.activeDropdown = null; this.dropdownTrigger?.focus(); return; }
            this.close();
        },
        dropdown(id, trigger) {
            this.dropdownTrigger = trigger;
            this.activeDropdown = this.activeDropdown === id ? null : id;
        },
        focusMenu(id, trigger) {
            this.dropdownTrigger = trigger; this.activeDropdown = id;
            this.$nextTick(() => trigger.closest('[data-nav-dropdown]')?.querySelector('[data-nav-dropdown-menu] a, [data-nav-dropdown-menu] button')?.focus());
        },
        moveMenu(event, direction) {
            const root = event.target.closest('[data-nav-dropdown-menu]');
            if (!root) { return; }
            const items = [...root.querySelectorAll('a[href], button:not([disabled])')];
            if (!items.length) { return; }
            event.preventDefault(); items[(items.indexOf(document.activeElement) + direction + items.length) % items.length]?.focus();
        },
        trap(event) {
            if (event.key !== 'Tab' || (!this.open && !this.logoutOpen)) { return; }
            const root = this.logoutOpen ? this.$refs.logoutDialog : this.$refs.panel;
            const items = [...(root?.querySelectorAll('a[href], button:not([disabled]), input, select, textarea') || [])].filter(el => el.getClientRects().length);
            if (!items.length) { event.preventDefault(); root?.focus(); return; }
            const first = items[0]; const last = items.at(-1);
            if (event.shiftKey && (document.activeElement === first || !items.includes(document.activeElement))) { event.preventDefault(); last.focus(); }
            else if (!event.shiftKey && (document.activeElement === last || !items.includes(document.activeElement))) { event.preventDefault(); first.focus(); }
        },
        requestLogout(trigger) {
            if (this.logoutOpen) { return; }
            this.logoutTrigger = trigger; this.logoutFromMenu = this.open; this.logoutSubmitting = false;
            this.logoutOpen = true; this.close(false);
            this.$nextTick(() => this.$refs.logoutDialog?.querySelector('button')?.focus());
        },
        cancelLogout() {
            if (this.logoutSubmitting) { return; }
            if (this.logoutFromMenu && !this.desktop) { this.open = true; this.broadcast(); }
            this.logoutOpen = false;
            this.syncLock();
            this.$nextTick(() => this.logoutTrigger?.focus());
        },
        containLogoutFocus(event) {
            if (this.logoutOpen && !this.$refs.logoutDialog?.contains(event.target)) {
                const dialog = this.$refs.logoutDialog;
                (dialog?.querySelector('button:not([disabled])') || dialog)?.focus();
            }
        },
        confirmLogout(form) {
            if (!this.logoutOpen || this.logoutSubmitting) { return; }
            this.logoutSubmitting = true;
            form.submit();
        },
        reset() {
            this.logoutSubmitting = false; this.logoutOpen = false; this.close(false);
        },
        async refreshUnread() {
            if (document.hidden) { return; }
            try {
                const response = await fetch(config.unreadUrl, { headers: { Accept: 'application/json' } });
                if (response.ok) { this.unread = Number((await response.json()).count) || 0; }
            } catch { /* Retain the last known count during a temporary network failure. */ }
        },
    };
}

export function navToggle(id) {
    return { expanded: false, toggle() { window.dispatchEvent(new CustomEvent('navigation-toggle', { detail: { id, trigger: this.$el } })); } };
}

import { test } from 'node:test';
import assert from 'node:assert/strict';

function classes(...initial) {
    const values = new Set(initial);
    return {
        add: (...names) => names.forEach(name => values.add(name)),
        remove: (...names) => names.forEach(name => values.delete(name)),
        contains: name => values.has(name),
        toggle: (name, enabled) => enabled ? values.add(name) : values.delete(name),
    };
}

let instance = 0;
async function setup({ reduced = false, supported = true, broken = false, width = 320 } = {}) {
    let directive;
    let intersection;
    let motionChange;
    const observed = new Set();
    const motion = { matches: reduced, addEventListener: (_, fn) => { motionChange = fn; } };
    globalThis.document = {
        body: { classList: classes('landing-page') },
        documentElement: { classList: classes() },
        querySelector: () => null,
    };
    globalThis.window = {
        innerWidth: width, innerHeight: 800,
        matchMedia: () => motion,
    };
    globalThis.getComputedStyle = () => ({ gridTemplateColumns: width < 640 ? '1fr 1fr' : '1fr 1fr 1fr 1fr 1fr 1fr' });
    if (supported) {
        globalThis.IntersectionObserver = window.IntersectionObserver = class {
            constructor(callback) {
                if (broken) { throw new Error('unavailable'); }
                intersection = callback;
            }
            observe(element) { observed.add(element); }
            unobserve(element) { observed.delete(element); }
        };
    }
    const { registerLandingEffects } = await import(`../../resources/js/pages/landing/landing.js?test=${++instance}`);
    registerLandingEffects({ directive: (_, callback) => { directive = callback; } });
    function element(top) {
        const style = new Map();
        const events = new Map();
        return {
            classList: classes(), parentElement: {},
            style: { setProperty: (key, value) => style.set(key, value), removeProperty: key => style.delete(key) },
            getBoundingClientRect: () => ({ top, bottom: top + 100 }),
            addEventListener: (name, callback) => events.set(name, callback),
            removeEventListener: name => events.delete(name),
            values: style, events,
        };
    }
    function apply(target, index = '') {
        let dispose;
        directive(target, { expression: String(index) }, { cleanup: callback => { dispose = callback; } });
        return dispose;
    }
    return { element, apply, observed, enter: target => intersection([{ target, isIntersecting: true }]), reduce: () => { motion.matches = true; motionChange(); } };
}

for (const width of [320, 768, 1024, 1440, 1920]) {
    test(`first viewport stays visible immediately at ${width}px`, async () => {
        const context = await setup({ width });
        const first = context.element(150);
        context.apply(first);
        assert.equal(first.classList.contains('reveal-pending'), false);
        assert.equal(context.observed.size, 0);
    });
}

test('below-fold items reveal once, stagger by column, and release animation hints', async () => {
    const context = await setup();
    const tile = context.element(1000);
    const dispose = context.apply(tile, 3);
    assert.equal(tile.classList.contains('reveal-pending'), true);
    assert.equal(tile.classList.contains('reveal-animating'), false);
    assert.equal(tile.values.get('--reveal-delay'), '60ms');
    context.enter(tile);
    assert.equal(tile.classList.contains('reveal-pending'), false);
    assert.equal(tile.classList.contains('reveal-animating'), true);
    assert.equal(context.observed.size, 0);
    dispose();
    assert.equal(tile.classList.contains('reveal-animating'), false);
    assert.equal(tile.values.size, 0);
});

test('reduced motion and observer failures leave everything visible', async () => {

        const context = await setup(options);
        const tile = context.element(1000);
        context.apply(tile);
        assert.equal(tile.classList.contains('reveal-pending'), false);
    }
});
